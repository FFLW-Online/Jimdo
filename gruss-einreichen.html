// ============================================================ // FFLW – Geheime Abschieds- & Videogruß-Seite ("Birgid") // Mit Direkt-Upload-Brücke für große 2-Minuten-Videos! // ============================================================ const ORDNER_ID = "1xQ-T7JcIhxOg9nmNGGqkcY0Fj1w8NMaD"; const GEHEIMWORT = "Birgid"; // Mit "d" am Ende function doGet(e) { const pw = (e && e.parameter && e.parameter.pw) ? e.parameter.pw.trim() : ""; if (pw.toLowerCase() !== GEHEIMWORT.toLowerCase()) { return jsonOut({ ok: false, error: "Falsches Geheimwort" }); } try { const folder = DriveApp.getFolderById(ORDNER_ID); const files = folder.getFiles(); const gruesse = []; const texteMap = {}; const allFiles = []; while (files.hasNext()) { const f = files.next(); allFiles.push(f); if (f.getName().endsWith(".txt")) { const baseName = f.getName().replace(/\.txt$/i, ""); try { texteMap[baseName] = JSON.parse(f.getBlob().getDataAsString("UTF-8")); } catch (err) { texteMap[baseName] = { absender: baseName, nachricht: f.getBlob().getDataAsString("UTF-8") }; } } } allFiles.forEach(file => { const mime = file.getMimeType(); if (mime.startsWith("video/") || mime.startsWith("image/")) { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); const rawName = file.getName(); const baseName = rawName.replace(/\.[^/.]+$/, ""); const meta = texteMap[baseName] || {}; const fallbackAbsender = baseName .replace(/^\d{8}_\d{6}_/, "") .replace(/^WhatsApp Video \d{4}-\d{2}-\d{2} at \d{2}\.\d{2}\.\d{2}/i, "Videogruß") .replace(/[_-]+/g, " ") .trim(); gruesse.push({ id: file.getId(), typ: mime.startsWith("video/") ? "video" : "bild", absender: meta.absender || fallbackAbsender || "Ein herzlicher Gruß", nachricht: meta.nachricht || "", datum: file.getDateCreated().getTime(), embedUrl: "https://drive.google.com/file/d/" + file.getId() + "/preview", downloadUrl: "https://drive.google.com/uc?export=download&id=" + file.getId() }); } }); gruesse.sort((a, b) => a.datum - b.datum); return jsonOut({ ok: true, gruesse: gruesse }); } catch (err) { return jsonOut({ ok: false, error: err.toString() }); } } function doPost(e) { try { const data = JSON.parse(e.postData.contents); const folder = DriveApp.getFolderById(ORDNER_ID); const absender = (data.absender || "Unbekannt").trim(); const nachricht = (data.nachricht || "").trim(); const zeitstempel = Utilities.formatDate(new Date(), "Europe/Berlin", "yyyyMMdd_HHmmss"); const sichererName = absender.replace(/[^a-zA-Z0-9äöüÄÖÜß -]/g, "").trim() || "Gruss"; const basisDateiname = zeitstempel + "_" + sichererName; // 1. Begleittext speichern const metaPayload = JSON.stringify({ absender: absender, nachricht: nachricht }); folder.createFile(basisDateiname + ".txt", metaPayload, MimeType.PLAIN_TEXT); // 2. Direkte Google-Drive-Upload-Session für große Videos (bis 2 Min. / 250 MB) starten const ext = (data.fileName && data.fileName.includes(".")) ? data.fileName.substring(data.fileName.lastIndexOf(".")) : ".mp4"; const targetName = basisDateiname + ext; const mimeType = data.mimeType || "video/mp4"; const metadata = { name: targetName, parents: [ORDNER_ID] }; const initUrl = "https://www.googleapis.com/upload/drive/v3/files?uploadType=resumable"; const response = UrlFetchApp.fetch(initUrl, { method: "post", contentType: "application/json; charset=UTF-8", headers: { Authorization: "Bearer " + ScriptApp.getOAuthToken(), "X-Upload-Content-Type": mimeType }, payload: JSON.stringify(metadata), muteHttpExceptions: true }); const uploadUrl = response.getHeaders()["Location"] || response.getHeaders()["location"]; if (!uploadUrl) { throw new Error("Upload-Sitzung konnte nicht gestartet werden: " + response.getContentText()); } return jsonOut({ ok: true, uploadUrl: uploadUrl, basisDateiname: basisDateiname }); } catch (err) { return jsonOut({ ok: false, error: err.toString() }); } } function jsonOut(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
